/* ---------------------------------------------------------
 * TEST 1: Ordenamiento de Burbuja (Bubble Sort)
 * Valida: Anidación fuerte de ciclos "para" e intercambios.
 * --------------------------------------------------------- */
funcion ordenarBurbuja($arr como arreglo [entero]) entrega arreglo [entero] >
    definir $n como entero con valor de longitud($arr).
    definir $temp como entero con valor de 0.
    definir $limiteI como entero con valor de $n - 1.

    para $i de 0 hasta $limiteI >
        definir $limiteJ como entero con valor de $n - $i - 2.

        para $j de 0 hasta $limiteJ >
            si ( $arr[$j] mayor que $arr[$j + 1] ) >
                // Intercambio de variables
                $temp con valor de $arr[$j].
                $arr[$j] con valor de $arr[$j + 1].
                $arr[$j + 1] con valor de $temp.
            <
        <
    <
    retornar $arr.
<


/* ---------------------------------------------------------
 * TEST 2: Ordenamiento por Inserción (Insertion Sort)
 * Valida: Ciclos "mientras" anidados, operadores lógicos
 *         compuestos y ruptura prematura (romper).
 * --------------------------------------------------------- */
funcion ordenarInsercion($arr como arreglo [entero]) entrega arreglo [entero] >
    definir $n como entero con valor de longitud($arr).
    definir $clave como entero con valor de 0.
    definir $j como entero con valor de 0.
    definir $limite como entero con valor de $n - 1.

    para $i de 1 hasta $limite >
        $clave con valor de $arr[$i].
        $j con valor de $i - 1.

        // Usamos "romper" para estresar el manejo de scopes
        mientras ( $j mayor o igual que 0 ) >
            si ( $arr[$j] mayor que $clave ) >
                $arr[$j + 1] con valor de $arr[$j].
                $j con valor de $j - 1.
            <
            sino >
                romper.
            <
        <
        $arr[$j + 1] con valor de $clave.
    <
    retornar $arr.
<


/* ---------------------------------------------------------
 * TEST 3: Ordenamiento por Selección (Selection Sort)
 * Valida: Asignaciones de índices, recálculo de expresiones
 *         en cabeceras de ciclos (inicio dinámico).
 * --------------------------------------------------------- */
funcion ordenarSeleccion($arr como arreglo [entero]) entrega arreglo [entero] >
    definir $n como entero con valor de longitud($arr).
    definir $min_idx como entero con valor de 0.
    definir $temp como entero con valor de 0.
    definir $limiteI como entero con valor de $n - 2.
    definir $limiteJ como entero con valor de $n - 1.

    para $i de 0 hasta $limiteI >
        $min_idx con valor de $i.
        definir $inicioJ como entero con valor de $i + 1.

        para $j de $inicioJ hasta $limiteJ >
            si ( $arr[$j] menor que $arr[$min_idx] ) >
                $min_idx con valor de $j.
            <
        <

        // Intercambio final del ciclo
        $temp con valor de $arr[$min_idx].
        $arr[$min_idx] con valor de $arr[$i].
        $arr[$i] con valor de $temp.
    <
    retornar $arr.
<
